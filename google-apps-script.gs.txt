// Atmos Weather Journal — Google Sheets storage endpoint
// SETUP
// 1. Create or open a Google Sheet.
// 2. Extensions → Apps Script; replace the editor contents with this file.
// 3. Deploy → New deployment → Web app.
// 4. Execute as: Me. Who has access: Anyone. Click Deploy.
// 5. Copy the /exec URL and paste it into Atmos using the “Sheets” button.

const SHEET_NAME = 'Weather Observations';

function doPost(e) {
  try {
    const payload = JSON.parse((e.postData && e.postData.contents) || '{}');
    const sheet = getObservationSheet_();
    sheet.appendRow([
      payload.timestamp || new Date().toISOString(),
      payload.location || '',
      payload.region || '',
      payload.latitude || '',
      payload.longitude || '',
      payload.temperature || '',
      payload.unit || 'C',
      payload.condition || '',
      payload.humidity || '',
      payload.wind_kmh || '',
      payload.pressure_hpa || '',
      payload.uv_index || ''
    ]);
    return output_({ ok: true, row: sheet.getLastRow() });
  } catch (error) {
    return output_({ ok: false, error: String(error) });
  }
}

function doGet() {
  const sheet = getObservationSheet_();
  const values = sheet.getDataRange().getDisplayValues();
  const headers = values.shift() || [];
  const observations = values.slice(-50).reverse().map(row =>
    headers.reduce((item, header, index) => {
      item[header] = row[index];
      return item;
    }, {})
  );
  return output_({ ok: true, observations: observations });
}

function getObservationSheet_() {
  const workbook = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = workbook.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = workbook.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'timestamp', 'location', 'region', 'latitude', 'longitude',
      'temperature', 'unit', 'condition', 'humidity', 'wind_kmh',
      'pressure_hpa', 'uv_index'
    ]);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, 12).setFontWeight('bold').setBackground('#173f52').setFontColor('#ffffff');
    sheet.autoResizeColumns(1, 12);
  }
  return sheet;
}

function output_(value) {
  return ContentService
    .createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}
